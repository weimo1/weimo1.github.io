```
#include <stdio.h>
#include <stdlib.h>

// 前置声明
typedef struct Shape Shape;

// 定义vtable结构
typedef struct {
    // 定义析构函数以外的虚函数，这里只有一个
    double (*area)(Shape*);
    // 析构函数需要定义成虚函数，这个是常识
    void (*destroy)(Shape*); 
} ShapeVTable;

// 基类结构
struct Shape {
    ShapeVTable* vtable;
    int x, y; // 公共数据，例如坐标
};

// 基类的通用函数，例如销毁函数
void shape_destroy(Shape* shape) {
    if (shape) {
        free(shape);
    }
}

// 基类的面积计算函数，可以定义为纯虚函数，即要求派生类实现
// 这里我们不需要实现，因为通过vtable调用派生类的函数

// 圆形派生类
typedef struct {
    Shape base; // 基类作为第一个成员，以实现继承
    double radius;
} Circle;

// 圆形的面积计算函数
double circle_area(Shape* shape) {
    Circle* circle = (Circle*)shape;
    return 3.14159 * circle->radius * circle->radius;
}

// 圆形的销毁函数
void circle_destroy(Shape* shape) {
    free(shape);
}

// 圆形的vtable
ShapeVTable circle_vtable = {
    .area = circle_area,
    .destroy = circle_destroy
};

// 圆形的构造函数，在这里面设置vtable
Circle* circle_create(int x, int y, double radius) {
    Circle* circle = malloc(sizeof(Circle));
    circle->base.vtable = &circle_vtable;
    circle->base.x = x;
    circle->base.y = y;
    circle->radius = radius;
    return circle;
}

// 矩形派生类
typedef struct {
    Shape base;
    double width, height;
} Rectangle;

// 矩形的面积计算函数
double rectangle_area(Shape* shape) {
    Rectangle* rect = (Rectangle*)shape;
    return rect->width * rect->height;
}

// 矩形的销毁函数
void rectangle_destroy(Shape* shape) {
    free(shape);
}

// 矩形的vtable
ShapeVTable rectangle_vtable = {
    .area = rectangle_area,
    .destroy = rectangle_destroy
};

// 矩形的构造函数
Rectangle* rectangle_create(int x, int y, double width, double height) {
    Rectangle* rect = malloc(sizeof(Rectangle));
    rect->base.vtable = &rectangle_vtable;
    rect->base.x = x;
    rect->base.y = y;
    rect->width = width;
    rect->height = height;
    return rect;
}

// 使用多态的函数
void print_area(Shape* shape) {
    double area = shape->vtable->area(shape);
    printf("Area: %f\n", area);
}

int main() {
    // 创建一个圆形和一个矩形
    Circle* circle = circle_create(0, 0, 5.0);
    Rectangle* rect = rectangle_create(0, 0, 4.0, 6.0);

    // 通过基类指针调用函数，实现多态
    print_area((Shape*)circle);
    print_area((Shape*)rect);

    // 销毁对象
    circle->base.vtable->destroy((Shape*)circle);
    rect->base.vtable->destroy((Shape*)rect);

    return 0;
}
```

目前主要用到的是查看虚表（vtable）的命令，是

```
g++ -fdump-lang-class[='filename'] source.cpp
```

注意到这个命令的名字和gcc版本有关：

Since GCC version 8.0 at least these two options were replaced:  
`-fdump-class-hierarchy` ---> `-fdump-lang-class`  
`-fdump-translation-unit` ---> `-fdump-lang-raw`

在linux下，不指定`filename`会直接生成`xxx.cpp.yyy.class`文件，里面就包含了所有类的虚表

虚函数：

虚表的构建在编译期 ，放在数据段

* 编译时生成结构
* 链接时分配地址
* 加载时进入内存

函数构造时，**先进入构造函数，分配内存后，先分配虚函数表**，然后再初始化列表，初始化变量，进入构造函数。

构造函数调用虚函数，初始化的时自己的虚函数表，所以调用的时自己声明的虚函数

虚函数表的顺序：

与你基类的顺序有关，先从基类的顺序排列，如果子类继承并且重写，则替换指向函数的指针，顺序不变，如果没有新加，则按你声明的顺序再基类后增加

1.构造父类；  
2. 将子类虚函数表地址给子类对象；  
3. 初始化列表进行初始化；  
4. 构造函数本体；

析构时也是一样，析构时调用虚函数也是自己的虚函数表，然后析构完成，析构父类类时，调用父类的虚函数表

### 链接器处理流程：

```
1. 符号收集阶段：
   ├─ 从目标文件收集所有函数符号
   ├─ 建立符号表
   └─ 解析符号依赖关系

2. 地址分配阶段：
   ├─ 为代码段分配基地址 (如 0x7FF759480000)
   ├─ 按照链接器算法排列函数
   └─ 计算每个函数的最终地址

3. 重定位阶段：
   ├─ 修正函数调用地址
   ├─ 构建最终vtable内容
   └─ 处理所有地址引用
```

### 4.2 链接器的函数排列策略：

从你的地址分布可以看出，链接器可能使用了以下策略：

```
1. 静态函数优先：staticFunction (00007FF7594810AF)
2. 虚函数聚集：B::C, A::D, B::PPP, A::p
3. 普通函数靠后：A::fun, ordinaryFunction  
4. 重写函数分组：B::p, B::D
```

```
程序启动时:
1. 全局变量初始化
2. 静态成员变量初始化  
3. vtable地址确定 (编译时已构建，运行时装载)
4. main函数开始执行
```

### 5.2 对象构造时的vtable设置：

```
// 创建B对象时：
B b_obj;

构造顺序:
1. 分配对象内存空间
2. 设置vptr指向A的vtable    (A构造函数执行前)
3. 执行A的构造函数
4. 设置vptr指向B的vtable    (B构造函数执行前)  
5. 执行B的构造函数
6. 对象构造完成
```

```
void analyzeVtableLayout() {
    cout << "\n=================== vtable布局分析 ===================" << endl;

    cout << "BaseA vtable地址: " << aVtable << endl;
    cout << "  [0] 析构函数: " << aVtable[0] << endl;
    cout << "  [1] virtualFunc: " << aVtable[1] << endl;
    cout << "  [2] show: " << aVtable[2] << endl;
    
    cout << "\n--- DerivedB对象vtable ---" << endl;

    cout << "  [0] 析构函数: " << bVtable[0] << endl;

}
```

```
        std::cout << "=== FinalDerived析构结束 ===\n\n";
    }

    virtual void virtualFunc() override {
        std::cout << "FinalDerived::virtualFunc()\n";
    }

    void showAllData() {
        std::cout << "Final data: " << final_data << std::endl;
        showLeftData();
        showRightData();
        showData();
    }
};

int main() {
    std::cout << "========== 开始创建FinalDerived对象 ==========\n\n";

    FinalDerived* obj = new FinalDerived();

    std::cout << "========== 对象创建完成，测试多态调用 ==========\n\n";

    // 通过不同指针类型调用虚函数
    std::cout << "通过FinalDerived*调用: ";
    obj->virtualFunc();

    std::cout << "通过LeftDerived*调用: ";
    static_cast<LeftDerived*>(obj)->virtualFunc();

    std::cout << "通过RightDerived*调用: ";
    static_cast<RightDerived*>(obj)->virtualFunc();

    std::cout << "通过VirtualBase*调用: ";
    static_cast<VirtualBase*>(obj)->virtualFunc();

    std::cout << "\n========== 显示所有数据 ==========\n";
    obj->showAllData();

    std::cout << "\n========== 开始析构对象 ==========\n\n";
    delete obj;

    std::cout << "========== 程序结束 ==========\n";

    return 0;
}
```

参考：[恋恋风辰官方博客](https://llfc.club/category?catid=225RaiVNI8pFDD5L4m807g7ZwmF#!aid/2CmcOeP6BZMbtNiglPTkbmnpb73)

### 虚函数表的核心原理

#### 1. 虚函数表的产生条件
- 当类中存在至少一个虚函数（包括虚析构函数）时，编译器会为该类生成一个虚函数表。
- 每个类（包括基类和派生类）有且仅有一个虚函数表。

#### 2. 虚函数指针（vptr）
- 每个对象实例的内存布局中会包含一个隐藏的指针（vptr），指向该对象所属类的虚函数表。
- vptr 通常位于对象内存布局的起始位置。

#### 3. 虚函数表的内容
- 虚函数表按顺序存储类中所有虚函数的地址（包括继承的虚函数）。
- 派生类会继承基类的虚函数表，并覆盖其中对应的虚函数地址。

### 虚函数表的内存布局

### 静态多态核心机制

#### 1. 模板

```c++

T add(T a, T b) { return a + b; }

int sum_int = add(3, 5);     // T = int
```

```c++
    void implementation() { std::cout << "Derived\n"; }

```

#### 3. 策略模式

```c++
template <typename AllocationPolicy>
```

### 静态多态 vs 动态多态对比

| 特性 | 静态多态 | 动态多态 |
| --- | --- | --- |
| 实现机制 | 模板、函数重载 | 虚函数、继承 |
| 解析时机 | 编译时 | 运行时 |
| 性能开销 | 无虚表查询，零开销 | 虚表跳转，轻微开销 |
| 代码膨胀 | 可能生成多个模板实例 | 单一虚表，无膨胀 |
| 灵活性 | 需编译时确定类型 | 支持运行时动态绑定 |

### 选择原则
- 静态多态：极致性能、编译时确定类型、避免虚函数开销
- 动态多态：运行时动态绑定、接口与实现分离、插件架构

## 一、核心概念

- **定义**：多态（Polymorphism）是面向对象三大特性之一，指同一接口对不同类型呈现不同行为。C++ 中有两种多态：**静态多态**（编译期，模板/重载/CRTP）和**动态多态**（运行期，虚函数/继承）
- **关键词**：虚函数表（vtable）、虚函数指针（vptr）、动态绑定、纯虚函数（`= 0`）、抽象基类、虚继承（virtual inheritance）、CRTP、thunk、RTTI、`final` 去虚化
- **适用场景/边界**：静态多态——极致性能要求、编译期类型已知、数值计算/模板库；动态多态——插件架构、运行时多类型切换、接口与实现分离、工厂模式

## 二、详细解析（≥200字）

- **原理拆解（三层递进）**：
  - **第一层——虚函数表（vtable）机制**：每个含虚函数的类在编译期生成一份虚表（.rodata 段），存储虚函数地址 + RTTI 信息。对象实例的首字段 vptr 指向所属类的虚表。调用 `p->func()` 编译为 `mov rax,[p]; mov rax,[rax+offset]; call rax`（三次内存访问）
  - **第二层——单继承中的覆盖与替换**：派生类重写虚函数时，其虚表中对应槽位被替换为派生类的函数地址。构造/析构期间 vptr 逐层切换：Base 构造时 vptr→Base::vtable，Derived 构造时 vptr→Derived::vtable。析构过程逆序切换。构造函数中调用虚函数只能调到当前类版本（不会派发到派生类）
  - **第三层——多继承与虚继承的复杂性**：多继承 `D : A, B, C` 中 D 拥有多张虚表（每基类一张），跨基类调用需要 this 指针调整（thunk）。虚继承 `D : virtual B` 解决菱形继承中多份基类子对象问题，通过 vbase 偏移表间接访问共享基类，额外引入间接层开销
- **关键数据结构/接口**：虚表结构（Itanium ABI）= `[type_info* RTTI][offset_to_top][&func1][&func2]...`；`sizeof(多态对象) ≥ sizeof(vptr) + 成员大小`（64位 vptr=8B）
- **关键公式**：虚调用开销 ≈ `2 次内存取（vptr + 函数指针）+ 1 次间接跳转 ≈ 5-10 cycles`（BTB 命中时 2-3 cycles）；非虚调用 ≈ `1-2 cycles`

## 四、进阶应用（≥500字）

### CRTP 静态多态——零开销接口

```c++
template<typename Derived>
struct Base {
    void interface() { static_cast<Derived*>(this)->impl(); }
};
struct Derived : Base<Derived> { void impl() { /* ... */ } };
// 编译后等价于 Derived::interface() { this->impl(); } —— 零虚表、零间接跳转
```

CRTP 的代价：每个 Derived 实例化一份 Base，代码膨胀；调试信息冗余；无法运行时统一存储不同类型到容器。用 `std::variant<D1, D2, D3>` + `std::visit` 可兼顾编译期多态与运行时选择。

### final 关键字去虚化

```c++
struct Base { virtual void f(); };
struct Derived final : Base { void f() override; };
// 编译器可推断：Derived* 调用 f() 必然是 Derived::f()，直接调用无虚表查
```

LTO（Link Time Optimization）跨编译单元去虚化：编译器知道完整类层次后可将虚调用转为直接调用，性能提升可超 20%。

### 多继承中的 this 调整（thunk）

```c++
struct A { virtual ~A(); int a_data; };
struct B { virtual ~B(); int b_data; };
struct D : A, B { virtual ~D(); int d_data; };
// D 内存: [A::vptr][A::a_data][B::vptr][B::b_data][D::d_data]
// B* p = new D; 时 p = this + sizeof(A)；delete p 时需 thunk 调整回原 this
```

### 虚继承的工程权衡

虚继承解决菱形继承的数据冗余，但付出代价：对象大小增加（vbase 偏移指针）、构造更复杂（最终派生类负责初始化虚基类）、内存访问多一次间接。**工程建议**：优先用组合替代多重继承；必须多继承时优先非虚继承 + 明确接口分离（纯虚接口不含数据成员）。

### 类型擦除——超越继承的多态

```c++
// std::function 用类型擦除替代虚函数继承，存储任意可调用对象而不要求继承
std::function<void()> f = [&]{ obj.foo(); };  // 无需 Obj 继承任何基类
```

类型擦除 = 外部多态（类型安全 + 值语义），虚函数 = 内部多态（侵入式继承）。现代 C++ 倾向前者（`std::function`, `std::any`, `std::variant` + visit）。

## 五、源码解析和实践感悟

### 1. 虚函数表的真实内存布局

```c++
// GCC Itanium ABI 下的对象布局
struct Base {
    void* vptr;      // ★ 偏移 0：指向虚表
    int base_data;    // 偏移 8
};
// sizeof(Base) == 16（含 vptr 8B + int 4B + padding 4B）

struct Derived : Base {
    int derived_data; // 偏移 16
};
// Derived 的 vptr 指向 Derived 的虚表（不是 Base 的）

// 虚表结构（简化）：
// VTable_Derived:
//   [0] type_info* RTTI
//   [1] offset_to_top (0 for primary base)
//   [2] &Derived::func1()     ← 覆盖基类虚函数
//   [3] &Derived::func2()     ← 新增虚函数
```

### 2. 动态多态的运行时开销

```c++
// 虚函数调用开销：三次内存访问
Base* p = get_object();
// 编译为：
// mov rax, [p]           ; ① 取 vptr
// mov rax, [rax + 8]     ; ② 查虚表取函数地址
// call rax                ; ③ 间接跳转（BTB 预测）

// 非虚函数调用：一次直接跳转
// call Base::func          ; 直接调用，无查表开销

// 开销量化：
// - 虚调用 ~5-10 cycles (BTB 命中时 2-3 cycles)
// - 直接调用 ~1-2 cycles
// - BTB miss 惩罚 ~15-20 cycles
```

### 3. CRTP 实现静态多态的零开销

```c++
template<typename Derived>
struct Base {
    void interface() {
        static_cast<Derived*>(this)->impl();  // 编译期绑定，无虚表查
    }
};

struct Derived : Base<Derived> {
    void impl() { /* 具体实现 */ }
};

// 编译后等价于：
void Derived::interface() { this->impl(); }  // 直接调用，完全内联
// 零虚表、零间接跳转、零 RTTI
```

### 实践经验

1. **虚函数不是银弹**：热路径中虚调用 + BTB miss 可能成为瓶颈，高频路径优先静态多态
2. **final 去虚化**：`class Derived final` 后编译器可将虚调用转为直接调用
3. **LTO 跨模块优化**：Link Time Optimization 可将跨 .cpp 的虚调用去虚化
4. **vptr 空间成本**：每个多态对象多 8 字节（64位），百万对象 = 8MB 额外开销
5. **A + B + C 多继承**：每个非主基类需要一个 adjustment thunk，构建/析构中 vptr 需多次切换
6. **CRTP 代价**：代码膨胀（每个 Derived 实例化一份 Base），调试信息冗余

## 六、面试准备

### Q&A（10题）

**Q1: 静态多态和动态多态的本质区别？**
A: 静态多态编译期绑定（模板/重载），无运行时开销；动态多态运行期绑定（虚函数/继承），有虚表查

**Q2: 虚函数表什么时候生成？**
A: 编译期生成，存在 .rodata 段。每个有虚函数的类一份虚表，包含函数指针 + RTTI

**Q3: vptr 什么时候被设置？**
A: 构造体第一条指令前，编译器插入 `mov [this], vtable_addr`。派生类构造中 vptr 逐层切换

**Q4: 虚函数调用开销有多大？**
A: 比直接调用多 2 次内存访问（取 vptr + 取函数指针）+ 1 次间接跳转。BTB 命中时 ~3-5 cycles 额外

**Q5: CRTP 为什么是零开销？**
A: 所有调用在编译期确定，`static_cast<Derived*>(this)` 只是 `this` 的类型转换，无运行时操作

**Q6: final 如何帮助去虚化？**
A: final 类保证无派生类，虚调用可通过分析减少虚表查；`final` 虚函数在该类中不可被覆盖

**Q7: 多继承下的 thunk 是什么？**
A: 调整 this 偏移的小函数。`Derived : A, B` 通过 B* 调用覆盖的函数时，this 需减去 B 的偏移

**Q8: 纯虚函数可以有实现吗？**
A: 可以。`virtual void f() = 0;` 后提供 `Base::f() { ... }`。派生类显式调用 `Base::f()`

**Q9: 什么时候用静态多态，什么时候用动态多态？**
A: 静态：性能关键、类型编译期已知；动态：插件架构、运行时多类型、接口统一

**Q10: RTTI 和虚函数表的关系？**
A: RTTI 存储在虚表的 `vptr[-1]` 位置（type_info*）。`typeid(*ptr)` 通过 vptr 获取 type_info

### 陷阱与反问（5个）

1. **陷阱**：构造函数中调用虚函数 — vptr 指向当前类虚表，不是派生类的，调不到派生类的重载
2. **反问**：全部用静态多态好吗？→ 不行，无法运行时切换类型，代码膨胀严重
3. **陷阱**：析构函数非虚 — `Base* p = new Derived; delete p;` 只调 Base 析构，资源泄漏
4. **反问**：模板能替代虚函数吗？→ 部分场景可以（如 `std::function` 用类型擦除替代继承），但不能完全替代
5. **陷阱**：CRTP 误用为 `Base<WrongDerived>` — 编译通过但运行时 static_cast 行为未定义

### 一句话答案（8个）

1. **静态多态**：模板/重载，编译期绑定，零开销
2. **动态多态**：继承/虚函数，运行期绑定，vtable 查
3. **vptr**：指向虚表的隐藏指针，每个多态对象第一个字段
4. **虚调用开销**：2 次内存取 + 1 次间接跳转
5. **CRTP**：编译期多态，static_cast 实现零开销接口
6. **final**：禁止继承/覆盖，帮助编译器去虚化
7. **thunk**：多继承中调整 this 偏移的胶水代码
8. **虚析构**：`delete` 基类指针的前提，确保派生类析构执行
