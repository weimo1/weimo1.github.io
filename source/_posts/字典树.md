#### P8306 【模板】字典树

```
#include<bits/stdc++.h>
using namespace std;
int trie[300004][100];
int cnt[3000004];
int idx;
string x[100005];

void insert(string x){
    int u=0;
    for(int i=0;i<x.size();i++){
        if(trie[u][x[i]-'0']==0)  trie[u][x[i]-'0']=++idx;
        u=trie[u][x[i]-'0'];
        cnt[u]++;
    }
}

int query(string x){
    int u=0;
    for(int i=0;i<x.size();i++){
        u=trie[u][x[i]-'0'];
        if(!cnt[u]) break;
    }
    return cnt[u];
}

int main(){
    int T;
    cin>>T;
    while(T--){
        int n,q,sum=0;
        cin>>n>>q;
        for(int i=1;i<=n;i++){
            cin>>x[i];
            sum+=(int)(x[i].size());
        }
        for(int i=0;i<=sum;i++){
            cnt[i]=0;
            for(int j=0;j<100;j++)
            trie[i][j]=0;
        }
        for(int i=1;i<=n;i++){
            insert(x[i]);
        }
        for(int i=1;i<=q;i++){
            cin>>x[i];
            cout<<query(x[i])<<endl;
        }
    }
}
```

#### P10471 最大异或对 The XOR Largest Pair：

```
#include<bits/stdc++.h>
using namespace std;
int a[200004];
int idx=0,trie[3200004][2];
int cnt[3200004];
void insert(int x){
  int u=0;
  for(int i=30;i>=0;i--){
    int c=(x>>i)&1;
    if(trie[u][c]==0) trie[u][c]=++idx;
    u=trie[u][c];
  }
}

int query(int x){
   int u=0,ans=0;
   for(int i=30;i>=0;i--){
    int c=(x>>i)&1;
    if(trie[u][c^1]!=0){
     u=trie[u][c^1];
     ans+=(1<<i);
    }
    else u=trie[u][c];
   }
   return ans;
}

int main(){
    int T;
    cin>>T;
    int ans=0;
    for(int i=1;i<=T;i++) {
        cin>>a[i];
        insert(a[i]);
        ans=max(ans,query(a[i]));
    }
    cout<<ans;
}

#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const int N = 2e5 + 7;
int a[N], idx; 

struct Node {
    int val;      
    int son[2];  
} t[N * 32];      

void insert(int x)
{
   int o=0;
   for(int i=30;i>=0;i--){
        int y=x>>i&1;
        int &u=t[o].son[y];
        if(!u) u=++idx;
        o=u;
   }
}



int getmax(int x)
{
    int o=0;
    int res=0;
    for(int i=30;i>=0;i--){
        int y=x>>i&1;
        int u=t[o].son[y];
        int v=t[o].son[!y];
        if(v){
            o=v;
            res|=1<<i;
        }else{
            o=u;
        }
    }
    return res;
}
int main()
{
    ios::sync_with_stdio(0), cin.tie(0), cout.tie(0);

    int n; 
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];


    insert(a[1]);
    int ans = 0;
    for (int i = 2; i <= n; i++) 
    {
        ans = max(ans, getmax(a[i])); 
        insert(a[i]);                 
    }

    cout << ans << '\n';
   
    return 0;
}
```
