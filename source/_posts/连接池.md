# 连接池

```
//
// Created by qiyu on 12/14/17.
//

#ifndef ORMPP_CONNECTION_POOL_HPP
#define ORMPP_CONNECTION_POOL_HPP

#include <chrono>
#include <condition_variable>
#include <deque>
#include <memory>
#include <mutex>
#include <string>
#include <tuple>

namespace ormpp {
template <typename DB>
class connection_pool {
 public:
  static connection_pool<DB> &instance() {
    static connection_pool<DB> instance;
    return instance;
  }

  // call_once
  void init(int maxsize, const std::string &host = "",
            const std::string &user = "", const std::string &passwd = "",
            const std::string &db = "", const std::optional<int> &timeout = {},
            const std::optional<int> &port = {}) {
    std::call_once(flag_, &connection_pool<DB>::init_impl, this, maxsize, host,
                   user, passwd, db, timeout, port);
  }

  std::shared_ptr<DB> get() {
    std::unique_lock<std::mutex> lock(mutex_);

    while (pool_.empty()) {
      if (condition_.wait_for(lock, std::chrono::seconds(3)) ==
          std::cv_status::timeout) {
        // timeout
        return nullptr;
      }
    }

    auto conn = pool_.front();
    pool_.pop_front();
    lock.unlock();

    if (conn == nullptr || !conn->ping()) {
      return create_connection();
    }

    // check timeout, idle time shuold less than 8 hours
    auto now = std::chrono::system_clock::now();
    auto last = conn->get_latest_operate_time();
    auto mins =
        std::chrono::duration_cast<std::chrono::minutes>(now - last).count();
    if ((mins - 6 * 60) > 0) {
      return create_connection();
    }

    conn->update_operate_time();
    return conn;
  }

  void return_back(std::shared_ptr<DB> conn) {
    if (conn == nullptr || conn->has_error()) {
      conn = create_connection();
    }
    std::unique_lock<std::mutex> lock(mutex_);
    pool_.push_back(conn);
    lock.unlock();
    condition_.notify_one();
  }

 private:
  void init_impl(int maxsize, const std::string &host, const std::string &user,
                 const std::string &passwd, const std::string &db,
                 const std::optional<int> &timeout,
                 const std::optional<int> &port) {
    args_ = std::make_tuple(host, user, passwd, db, timeout, port);

    for (int i = 0; i < maxsize; ++i) {
      auto conn = std::make_shared<DB>();
      if (conn->connect(args_)) {
        pool_.push_back(conn);
      }
      else {
        throw std::invalid_argument("init failed");
      }
    }
  }

  auto create_connection() {
    auto conn = std::make_shared<DB>();
    return conn->connect(args_) ? conn : nullptr;
  }

  connection_pool() = default;
  ~connection_pool() = default;
  connection_pool(const connection_pool &) = delete;
  connection_pool &operator=(const connection_pool &) = delete;

  std::mutex mutex_;
  std::once_flag flag_;
  std::condition_variable condition_;
  std::deque<std::shared_ptr<DB>> pool_;
  std::tuple<std::string, std::string, std::string, std::string,
             std::optional<int>, std::optional<int>>
      args_;
};

template <typename DB>
struct conn_guard {
  conn_guard(std::shared_ptr<DB> con) : conn_(con) {}
  ~conn_guard() { connection_pool<DB>::instance().return_back(conn_.lock()); }

 private:
  std::weak_ptr<DB> conn_;
};
}  // namespace ormpp

#endif  // ORMPP_CONNECTION_POOL_HPP
```

## 五、源码解析和实践感悟

### 5.1 源码解析

#### 双重保险：call_once 防止并发初始化

```cpp
void init(int maxsize, ...) {
    std::call_once(flag_, &connection_pool<DB>::init_impl, this, maxsize, ...);
}
```

连接池初始化只需执行一次，`call_once` 保证即使在多线程同时首次 `get()` 时也只会有一个线程执行 `init_impl`。注意 `flag_` 是实例成员（非静态），每个池实例独立控制初始化。

#### get() 的超时等待机制

```cpp
std::shared_ptr<DB> get() {
    std::unique_lock<std::mutex> lock(mutex_);
    while (pool_.empty()) {
        if (condition_.wait_for(lock, std::chrono::seconds(3)) ==
            std::cv_status::timeout) {
            return nullptr;  // 超时返回空，避免无限阻塞
        }
    }
    // ...
}
```

**关键细节**：① `wait_for` 而非 `wait`——给定上限防止永久阻塞；② `while` 循环而非 `if`——防止虚假唤醒导致取空队列；③ 返回 `nullptr` 而非抛异常——让调用方自行决策。

#### return_back 的健康检查与自动修复

```cpp
void return_back(std::shared_ptr<DB> conn) {
    if (conn == nullptr || conn->has_error()) {
        conn = create_connection();  // 坏连接被替换
    }
    pool_.push_back(conn);
    condition_.notify_one();  // 唤醒等待的 get() 调用
}
```

归还时自动检查连接健康状态——坏连接被新连接替换后再入池。这样池中始终只有健康连接，避免下次 `get()` 取到坏连接。

### 5.2 实践经验

1. **连接池初始大小一定是 0 吗？** 否。建议启动时预热（如创建 5-10 个连接），避免首个请求因建立连接而超时
2. **归还比获取更重要**：连接泄漏（忘记 return_back）是生产中最常见的池故障——RAII guard 是根治方案
3. **ping() 的实现成本**：每次 `get()` 都 ping 对 MySQL 是轻量操作（COM_PING 几乎无开销），但高频场景可改为间隔 ping
4. **空闲超时别设太激进**：`wait_for` 设 3 秒超时合理（TCP 连接超时通常 30 秒+），太短会导致频繁创建新连接
5. **单例 + 连接池 = 双刃剑**：全局唯一方便但不利于多数据库实例、不利于测试。单元测试中可注入 mock 池
6. **监控每次 get 耗时**：若平均耗时显著上升 → 池大小不够或数据库响应变慢

## 六、面试准备

### 6.1 面试 Q&A

**Q1: 连接池的核心数据结构是什么？**

`std::deque<std::shared_ptr<DB>> pool_`——双端队列存储可用连接。`deque` 而非 `stack`/`queue` 可支持 LIFO 或 FIFO 调度。`shared_ptr` 保证连接在外借期间不被销毁。

**Q2: 为什么用 shared_ptr 管理连接？**

① 外借连接时 `shared_ptr` 计数 +1，连接不会在池析构时被销毁；② `conn_guard` 用 `weak_ptr` 观察连接是否仍存活；③ 多线程间安全传递所有权。

**Q3: 连接池的 call_once 为什么是成员变量而非静态？**

每个 `connection_pool<DB>` 实例独立——不同 DB 类型（MySQL、PostgreSQL）各有一个池，各自的 `init` 独立调用一次。作为实例成员允许运行时创建多个不同类型的池。

**Q4: 归还连接时如果池已满（超过初始 maxsize）怎么办？**

此实现中 `return_back` 无条件 push_back，未设上限。若需上限控制，应在 push 前检查 `pool_.size() >= maxsize` 并丢弃多余连接。

**Q5: 连接池获取超时后如何重试？**

策略：① 简单重试 N 次（不推荐——可能放大故障）；② 重试 + 退避（exponential backoff）；③ 快速失败 + 上层熔断降级。推荐方案③——让调用方感知故障并降级（如从缓存读）。

**Q6: 为什么 deque 而不是 vector 或 list？**

`deque` 两端 O(1) 操作——从队首取（front + pop_front），队尾还（push_back）。`vector` 在队首删除 O(n)，`list` 内存碎片化。`deque` 是连接池的最佳容器。

**Q7: 多线程环境下 pool_ 的线程安全如何保证？**

`std::mutex mutex_` + `std::condition_variable condition_` 经典组合。`get()` 在等待期间 `unique_lock` 允许 `wait_for` 原子地 unlock → sleep → lock，避免死锁。

**Q8: 连接池如何与 ORM 框架（如 ORMPP）集成？**

ORM 框架通过 `conn_guard` 绑定连接生命周期：
```cpp
auto guard = conn_guard(pool.get());  // 获取连接
auto result = guard->query("SELECT ...");  // 使用连接
// guard 析构自动归还连接
```
完全透明，业务代码无需关心获取/归还细节。

### 6.2 常见陷阱与面试反问

1. **陷阱**：`pool_.front()` 后忘记 `pop_front()` → 同一连接被多次分配。**修复**：确保 `front()` + `pop_front()` 成对出现。

2. **陷阱**：`notify_one()` 放在 `unlock()` 前 → 被通知线程被唤醒后立即阻塞在 lock 上（"hurry up and wait"）。**改进**：`unlock()` 后再 `notify_one()`（本代码已正确处理）。

3. **陷阱**：`condition_.wait_for` 返回后未检查返回值 → 可能使用 `nullptr` 连接。**修复**：检查超时返回 `nullptr`。

4. **反问**：「如果数据库连接数达到上限，你的连接池如何处理？」希望听到：① 设 max_size 限制池大小；② `get()` 超时返回空；③ 监控告警；④ 业务侧限流降级。

5. **反问**：「在多实例部署（K8s）下，每个 Pod 的连接池大小如何设置？」希望听到：总连接数 = 单 Pod 池大小 × Pod 数 ≤ DB max_connections。Pod 数变化时需动态调整或使用 ProxySQL/PgBouncer 等连接池中间件。
